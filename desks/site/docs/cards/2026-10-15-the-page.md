## Gate — run it; done means this passes
Save the script below as `gate_page.py` in your scratch folder, then, from the worktree's top folder:
`python <scratch>/gate_page.py desks/site/public <scratch>/serve` → `GATE PASS` (exit 0). Any failure prints `GATE FAIL` and one line per failure (exit 1).

```python
import functools, http.server, os, re, shutil, subprocess, sys, threading
pub, serve = sys.argv[1], sys.argv[2]
fails = []
html = open(os.path.join(pub, "index.html"), encoding="utf-8").read()
text = ""
for root, _, files in os.walk(pub):
    for f in files:
        if f.endswith((".html", ".css", ".js")):
            text += open(os.path.join(root, f), encoding="utf-8").read()
ids = ["name", "top", "film", "how", "founders", "organisations", "trust", "contact"]
pos = [html.find('id="%s"' % i) for i in ids]
if -1 in pos or pos != sorted(pos): fails.append("sections missing or out of order: %s" % dict(zip(ids, pos)))
if re.search(r"googleapis|gstatic|https?://[^\"')\s]*\.(woff2?|css|js)\b", text): fails.append("loads something from another site")
woff = [f for f in os.listdir(os.path.join(pub, "fonts")) if f.endswith(".woff2")] if os.path.isdir(os.path.join(pub, "fonts")) else []
if not woff or not os.path.isfile(os.path.join(pub, "fonts", "OFL.txt")): fails.append("no Inter woff2 and OFL.txt in fonts/")
if not re.search(r"@font-face[^}]*Inter[^}]*fonts/[^}]*\.woff2", text, re.S): fails.append("no @font-face for Inter from fonts/")
if not re.search(r"aspect-ratio:\s*16\s*/\s*9", text): fails.append("no 16:9 frame")
if html.count("mailto:rhys@foundrynorthstudios.co.uk") < 1: fails.append("no email button to rhys@")
if 'name="viewport"' not in html: fails.append("no viewport meta")
for hexv in ("#e7ede6", "#0e1512", "#e9b4b7", "#c99c85"):
    if hexv not in text.lower(): fails.append("palette colour missing: " + hexv)
vis = re.sub(r"<(style|script)\b.*?</\1>", "", html, flags=re.S)
if "\u2014" in vis or re.search(r">[^<]*![^<]*<", vis): fails.append("an em dash or exclamation mark in the page")
shutil.rmtree(serve, ignore_errors=True); shutil.copytree(pub, serve)
open(os.path.join(serve, "_wrap.html"), "w").write('<body><iframe id="f" src="index.html" style="width:360px;height:800px;border:0"></iframe><script>f.onload=()=>document.body.setAttribute("data-sw", f.contentDocument.documentElement.scrollWidth)</script></body>')
srv = http.server.ThreadingHTTPServer(("127.0.0.1", 0), functools.partial(http.server.SimpleHTTPRequestHandler, directory=serve))
threading.Thread(target=srv.serve_forever, daemon=True).start()
chrome = r"C:\Program Files\Google\Chrome\Application\chrome.exe"
dom = subprocess.run([chrome, "--headless=new", "--disable-gpu", "--virtual-time-budget=5000", "--dump-dom", "http://127.0.0.1:%d/_wrap.html" % srv.server_address[1]], capture_output=True, text=True, timeout=60).stdout
srv.shutdown()
sw = re.search(r'data-sw="(\d+)"', dom)
if not sw or int(sw.group(1)) > 360: fails.append("sideways scroll at 360 px: scrollWidth %s" % (sw.group(1) if sw else "unread"))
print("GATE PASS" if not fails else "GATE FAIL\n" + "\n".join(fails)); sys.exit(1 if fails else 0)
```
