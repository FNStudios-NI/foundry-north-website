## Gate — run it; done means this passes
Save the script below as `gate_words.py` in your scratch folder; from the worktree's top folder:
`python <scratch>/gate_words.py` → `GATE PASS` (exit 0), and `python tools/page_check.py docs/pages/2026-10-15-words-sheet-1.html --project .` → its pass line, and `python tools/people_check.py` → its pass line.

```python
import html.parser, re, sys
page = open("desks/site/public/index.html", encoding="utf-8").read()
words = open("desks/site/docs/words.md", encoding="utf-8").read()
class P(html.parser.HTMLParser):
    def __init__(s): super().__init__(); s.out = []; s.skip = 0
    def handle_starttag(s, tag, a):
        a = dict(a)
        if tag in ("style", "script"): s.skip += 1
        for k in ("aria-label", "alt"):
            if a.get(k): s.out.append(a[k])
        if tag == "meta" and (a.get("name") == "description" or (a.get("property") or "") in ("og:title", "og:description")): s.out.append(a.get("content") or "")
    def handle_endtag(s, tag):
        if tag in ("style", "script"): s.skip -= 1
    def handle_data(s, d):
        if not s.skip and d.strip(): s.out.append(" ".join(d.split()))
p = P(); p.feed(page)
quoted = set(" ".join(q.split()) for q in re.findall(r"`([^`]+)`", words))
fails = ["not in words.md: " + t for t in p.out if t not in quoted and not any(t in q for q in quoted)]
banned = r"\u2014|!|\b(forge|forged|alchemy|transformation|leverage|unlock|scale|empower|deliver|partner with|authority engine|ai-powered)\b"
for t in p.out:
    if re.search(banned, t, re.I): fails.append("banned: " + t)
for ruled in ("An engine that understands you better every month.", "your own mini studio", "Built by filmmakers. Fifteen years of craft.", "The last cut is human. Every time.", "mailto:rhys@foundrynorthstudios.co.uk"):
    if ruled.lower() not in page.lower(): fails.append("ruled line missing from the page: " + ruled)
n = len(re.findall(r"^\s*[-*]?\s*\**W\d+", words, re.M))
if n < len(set(p.out)) * 0.5: fails.append("too few numbered W lines: %d" % n)
print("GATE PASS" if not fails else "GATE FAIL\n" + "\n".join(fails)); sys.exit(1 if fails else 0)
```
