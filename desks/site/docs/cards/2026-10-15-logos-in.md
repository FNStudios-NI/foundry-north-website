```python
import os, re, sys
pub = "desks/site/public"
h = open(os.path.join(pub, "index.html"), encoding="utf-8").read()
sec = re.search(r'<section[^>]*id="logos".*?</section>', h, re.S)
fails = []
if not sec: fails.append("no logos section")
else:
    s = sec.group(0)
    if "standin" in s: fails.append("a stand-in is still in the banner")
    srcs = set(re.findall(r'src="logos/([^"]+\.png)"', s))
    if len(srcs) != 18: fails.append("expected 18 distinct logos, found %d" % len(srcs))
    for f in srcs:
        if not os.path.isfile(os.path.join(pub, "logos", f)): fails.append("missing file " + f)
    named = [a for a in re.findall(r'<img[^>]*alt="([^"]*)"', s) if a]
    if len(set(named)) < 18: fails.append("fewer than 18 named alts: %d" % len(set(named)))
left = [f for f in os.listdir(os.path.join(pub, "logos")) if f.startswith("standin")]
if left: fails.append("stand-in files left: %s" % left)
print("GATE PASS" if not fails else "GATE FAIL\n" + "\n".join(fails)); sys.exit(1 if fails else 0)
```
